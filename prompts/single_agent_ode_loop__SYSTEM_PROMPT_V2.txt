  import json
  import numpy as np
  from scipy.integrate import solve_ivp

  def rhs(t, y):
      x0, x1 = y
      d0 = 1.0*x0 - 1.0*x0*x1   # <-- replace with your symbolic_form[0]
      d1 = 1.0*x0*x1 - 1.0*x1   # <-- replace with your symbolic_form[1]
      return [d0, d1]

  y0 = [1.5, 1.0]
  t_eval = np.linspace(0, 25, 251)
  sol = solve_ivp(rhs, (t_eval[0], t_eval[-1]), y0, t_eval=t_eval, method="RK45")

  print(json.dumps({
      "integrated_at": sol.t.tolist(),
      "integrated_vals": sol.y.tolist(),
  }))
